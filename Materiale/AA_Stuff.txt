operatorul insert care primeste un intreg si un arbore echilibrat
si introduce un nou nod, tinand arborele echilibrat

Arbore sa fie echilibrat:
adancime(s) = adancime(d)
adancime(s) - 1 = adancime(d)

depth : Btree -> Int
depth(Empty) = 0
depth(Node(v, s, d)) = max(depth(s), depth(d)) + 1

insert : Intreg x BTree -> BTree
insert(v, Empty) = Node(v, Empty, Empty)
insert(v, Node(e, s, d)) = 
    if depth(s) - 1 == depth(d):
        Node(e, s, insert(v, d))
    else: 
        Node(e, insert(v, s), d)

add(Dec(y), x) = simplify(Dec(add(y, x)))

simplify Succ(Dec(x)) = x
simplify Dec(Suc(x)) = x
simplify x = x // daca x != Succ(Dec(...)) si x != Dec(Suc(...))

Succ(Dec(Zero)) = Zero

pop : Stack -> Stack
pop (EmptyS) = EmptyS
pop (push(v, s)) = s

peek : Stack \ EmptyS -> Int

Queue

EmptyQ :-> Queue
Enqueue : Int x Queue -> Queue

dequeue : Queue\EmptyQ -> Queue
dequeue (Enqueue(e, EmptyQ)) = EmptyQ
dequeue (Enqueue(e, q)) = Enqueue(e, dequeue(q))
    // q nu este de forma EmptyQ

peek : Queue\EmptyQ -> Int
peek (Enqueue(e, EmptyQ)) = e
peek (Enqueue(e, q)) = peek(q)
